    // ========== ESCAPE HTML ==========
    function escapeHTML(str) {
      return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#039;');
    }

    function ativarAba(aba) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
      document.getElementById('tab-' + aba).classList.add('active');
      document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
      document.querySelector(`.nav-item[data-tab="${aba}"]`).classList.add('active');
      if (aba === 'pedidos') renderizarPedidos();
      if (aba === 'consumo') renderizarClientesConsumo();
      if (aba === 'credito') renderizarClientesCredito();
      if (aba === 'clientes') renderizarClientes();
      if (aba === 'produtos') renderizarProdutos();
      lucide.createIcons();
    }

    // ========== GITHUB SYNC ==========
    function setSyncStatus(syncing) {
      const indicator = document.getElementById('syncIndicator');
      const label = document.getElementById('syncLabel');
      if (syncing) {
        indicator.classList.add('syncing');
        label.textContent = 'Sincronizando...';
      } else {
        indicator.classList.remove('syncing');
        label.textContent = 'Sincronizado';
      }
    }

    async function syncWithGitHub() {
      if (!config.token || !config.repo) {
        showToast('Configure GitHub nas Configurações', 'error');
        return;
      }
      setSyncStatus(true);
      try {
        const url = `https://api.github.com/repos/${config.repo}/contents/${config.file}`;
        const headers = {
          'Authorization': 'token ' + config.token,
          'Accept': 'application/vnd.github.v3+json'
        };
        // Buscar SHA atual
        let sha = null;
        try {
          const getRes = await fetch(url, { headers });
          if (getRes.ok) {
            const getData = await getRes.json();
            sha = getData.sha;
          }
        } catch (e) { }
        const content = btoa(unescape(encodeURIComponent(JSON.stringify(DB, null, 2))));
        const body = {
          message: 'Atualização Caderninho Digital Pro',
          content: content
        };
        if (sha) body.sha = sha;
        const putRes = await fetch(url, {
          method: 'PUT',
          headers: headers,
          body: JSON.stringify(body)
        });
        if (!putRes.ok) {
          const err = await putRes.json();
          throw new Error(err.message || 'Erro ao sincronizar');
        }
        showToast('Dados sincronizados com sucesso!', 'success');
      } catch (err) {
        showToast('Erro na sincronização: ' + err.message, 'error');
      } finally {
        setSyncStatus(false);
      }
    }

    // ========== CONFIG ==========
    function carregarConfig() {
      try {
        const saved = localStorage.getItem('caderninhoConfig');
        if (saved) {
          config = JSON.parse(saved);
          document.getElementById('githubToken').value = config.token || '';
          document.getElementById('githubRepo').value = config.repo || '';
          document.getElementById('githubFile').value = config.file || 'data.json';
        }
      } catch (e) {}
    }

    function salvarConfig() {
      config.token = document.getElementById('githubToken').value;
      config.repo = document.getElementById('githubRepo').value;
      config.file = document.getElementById('githubFile').value || 'data.json';
      localStorage.setItem('caderninhoConfig', JSON.stringify(config));
      showToast('Configuração salva!', 'success');
    }

    // ========== DADOS ==========
    function carregarDados() {
      try {
        const saved = localStorage.getItem('caderninhoDB');
        if (saved) {
          const parsed = JSON.parse(saved);
          if (parsed.clients) DB.clients = parsed.clients;
          if (parsed.products) DB.products = parsed.products;
          if (parsed.orders) DB.orders = parsed.orders;
        }
      } catch (e) {}
      renderizarTudo();
    }

    function salvarDados() {
      localStorage.setItem('caderninhoDB', JSON.stringify(DB));
      // Atualizar interface
      renderizarTudo();
    }

    }

    // ========== CLIENTES ==========
    function renderizarClientes() {
      const search = document.getElementById('clienteSearch').value.toLowerCase();
      const lista = document.getElementById('clienteLista');
      const filtered = DB.clients.filter(c => c.name.toLowerCase().includes(search));
      if (filtered.length === 0) {
        lista.innerHTML = '<p class="text-muted text-center">Nenhum cliente encontrado.</p>';
        return;
      }
      lista.innerHTML = filtered.map(c => `
        <div class="list-item">
          <div class="info">
            <strong>${escapeHTML(c.name)}</strong>
            <div class="text-xs text-muted">Saldo: R$ ${c.balance.toFixed(2)}</div>
          </div>
          <div class="actions">
            <button class="btn btn-sm btn-secondary" onclick="editarCliente('${c.id}')">
              <i data-lucide="pencil" style="width:14px;height:14px;"></i>
            </button>
            <button class="btn btn-sm btn-danger" onclick="excluirCliente('${c.id}')">
              <i data-lucide="trash-2" style="width:14px;height:14px;"></i>
            </button>
          </div>
        </div>
      `).join('');
      lucide.createIcons();
    }

    function salvarCliente(id = null) {
      const nome = document.getElementById('clienteNomeModal').value.trim();
      const saldo = parseFloat(document.getElementById('clienteSaldoModal').value);
      if (!nome) {
        showToast('Nome do cliente é obrigatório.', 'error');
        return;
      }
      if (isNaN(saldo) || saldo < 0) {
        showToast('Saldo inválido.', 'error');
        return;
      }
      if (id) {
        const idx = DB.clients.findIndex(c => c.id === id);
        if (idx !== -1) {
          DB.clients[idx].name = nome;
          DB.clients[idx].balance = saldo;
        }
      } else {
        const newId = Date.now().toString(36) + Math.random().toString(36).substr(2, 5);
        DB.clients.push({ id: newId, name: nome, balance: saldo });
      }
      salvarDados();

    function salvarProduto(id = null) {
      const nome = document.getElementById('produtoNomeModal').value.trim();
      const preco = parseFloat(document.getElementById('produtoPrecoModal').value);
      if (!nome) {
        showToast('Nome do produto é obrigatório.', 'error');
        return;
      }
      if (isNaN(preco) || preco < 0) {
        showToast('Preço inválido (não pode ser negativo).', 'error');
        return;
      }
      if (id) {
        const idx = DB.products.findIndex(p => p.id === id);
        if (idx !== -1) {
          DB.products[idx].name = nome;
          DB.products[idx].price = preco;
        }
      } else {
        const newId = Date.now().toString(36) + Math.random().toString(36).substr(2, 5);
        DB.products.push({ id: newId, name: nome, price: preco });
      }
      salvarDados();
      fecharModal();
      // Atualizar select de produtos
      atualizarSelectProdutos();
      showToast(id ? 'Produto atualizado!' : 'Produto criado!', 'success');
    }

    function editarProduto(id) {

    function addCarrinho() {
      const prodId = document.getElementById('consumoProdutoSelect').value;
      const qtd = parseInt(document.getElementById('consumoQtd').value) || 1;
      if (!prodId) {
        showToast('Selecione um produto.', 'error');
        return;
      }
      if (qtd < 1 || qtd > 999) {
        showToast('Quantidade inválida (1-999).', 'error');
        return;
      }
      const produto = DB.products.find(p => p.id === prodId);
      if (!produto) {
        showToast('Produto não encontrado.', 'error');
        return;
      }
      const existente = consumoState.carrinho.find(item => item.productId === prodId);
      if (existente) {
        existente.qty += qtd;
      } else {
        consumoState.carrinho.push({ productId: prodId, name: produto.name, price: produto.price, qty: qtd });
      }
      atualizarCarrinho();
      showToast('Item adicionado ao carrinho!', 'success');
    }

    function atualizarCarrinho() {
      const container = document.getElementById('carrinhoItems');
      const totalSpan = document.getElementById('carrinhoTotal');
      const countSpan = document.getElementById('cartItemsCount');
      if (consumoState.carrinho.length === 0) {
        container.innerHTML = '<p class="text-muted text-xs">Carrinho vazio.</p>';
        totalSpan.textContent = 'Total: R$ 0,00';
        countSpan.textContent = '0 itens';
        return;
      }
      let total = 0;
      let qtdTotal = 0;
      container.innerHTML = consumoState.carrinho.map((item, index) => {
        const subtotal = item.price * item.qty;
        total += subtotal;
        qtdTotal += item.qty;
        return `
          <div class="carrinho-item">
            <div>
              <strong class="text-sm">${escapeHTML(item.name)}</strong>
              <div class="text-xs text-muted">R$ ${item.price.toFixed(2)} cada</div>
            </div>
            <div class="flex items-center gap-2">
              <div class="qty-controls">
                <button onclick="alterarQtdCarrinho(${index}, -1)">-</button>
                <span>${item.qty}</span>
                <button onclick="alterarQtdCarrinho(${index}, 1)">+</button>
              </div>
              <button class="btn btn-sm btn-danger" onclick="removerItemCarrinho(${index})">
                <i data-lucide="x" style="width:14px;height:14px;"></i>
              </button>
            </div>
          </div>
        `;
      }).join('');
      totalSpan.textContent = `Total: R$ ${total.toFixed(2)}`;
      countSpan.textContent = `${qtdTotal} itens`;
      lucide.createIcons();
    }

    function alterarQtdCarrinho(index, delta) {
      const item = consumoState.carrinho[index];
      if (!item) return;
      item.qty += delta;
      if (item.qty < 1) item.qty = 1;
      atualizarCarrinho();
    }

    function removerItemCarrinho(index) {
      consumoState.carrinho.splice(index, 1);
      atualizarCarrinho();
    }

    function confirmarPedido() {
      if (!consumoState.selectedClient) {
        showToast('Selecione um cliente.', 'error');
        return;
      }
      if (consumoState.carrinho.length === 0) {
        showToast('Carrinho vazio.', 'error');
        return;
      }
      let totalDebito = 0;
      const items = consumoState.carrinho.map(item => {
        totalDebito += item.price * item.qty;
        return {
          productId: item.productId,
          name: item.name,
          price: item.price,
          qty: item.qty
        };
      });
      // Debita do cliente (permite saldo negativo)
      const clienteIdx = DB.clients.findIndex(c => c.id === consumoState.selectedClient.id);
      if (clienteIdx === -1) {
        showToast('Cliente não encontrado.', 'error');
        return;
      }
      DB.clients[clienteIdx].balance -= totalDebito;
      // Registra pedido
      const pedido = {
        id: Date.now().toString(36) + Math.random().toString(36).substr(2, 5),
        clientId: consumoState.selectedClient.id,
        clientName: consumoState.selectedClient.name,
        items: items,
        total: totalDebito,
        date: new Date().toISOString()
      };
      DB.orders.push(pedido);
      // Limpa carrinho
      consumoState.carrinho = [];
      salvarDados();
      atualizarCarrinho();
      showToast('Pedido confirmado com sucesso!', 'success');
      // Atualizar interface de consumo
      document.getElementById('consumoClienteNome').textContent = DB.clients[clienteIdx].name + ' (Saldo: R$ ' + DB.clients[clienteIdx].balance.toFixed(2) + ')';
    }

    function creditarDebitar() {
      if (!creditoState.selectedClient) {
        showToast('Selecione um cliente.', 'error');
        return;
      }
      const tipo = document.getElementById('creditoTipo').value;
      const valor = parseFloat(document.getElementById('creditoValor').value);
      if (isNaN(valor) || valor <= 0) {
        showToast('Valor inválido.', 'error');
        return;
      }
      const clienteIdx = DB.clients.findIndex(c => c.id === creditoState.selectedClient.id);
      if (clienteIdx === -1) {
        showToast('Cliente não encontrado.', 'error');
        return;
      }
      if (tipo === 'credito') {
        DB.clients[clienteIdx].balance += valor;
      } else {
        DB.clients[clienteIdx].balance -= valor; // permite negativo
      }
      salvarDados();
      creditoState.selectedClient = DB.clients[clienteIdx];
      document.getElementById('creditoSaldoAtual').textContent = 'Saldo: R$ ' + DB.clients[clienteIdx].balance.toFixed(2);
      document.getElementById('creditoValor').value = '';
      showToast('Operação realizada!', 'success');
    }
